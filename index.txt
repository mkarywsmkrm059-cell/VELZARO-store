const CART_KEY = "velzaro_cart_v1";
let cart = JSON.parse(localStorage.getItem(CART_KEY)) || [];
let activeCategory = "all";

const productsGrid = document.getElementById("productsGrid");
const cartItemsContainer = document.getElementById("cartItems");
const cartCount = document.getElementById("cartCount");
const cartTotal = document.getElementById("cartTotal");
const checkoutBtn = document.getElementById("checkoutBtn");
const toast = document.getElementById("toast");

function saveCart() {
localStorage.setItem(CART_KEY, JSON.stringify(cart));
}

function showToast(msg) {
toast.textContent = msg;
toast.classList.add("show");
setTimeout(() => toast.classList.remove("show"), 2000);
}

function renderProducts() {
const filtered = activeCategory === "all"
? perfumes
: perfumes.filter(p => p.category === activeCategory);

productsGrid.innerHTML = filtered.map(item => &lt;article class="card"&gt; &lt;div class="bottle-visual"&gt; &lt;span class="category-tag"&gt;${item.categoryLabel} • ${item.size}&lt;/span&gt; &lt;div class="bottle-icon"&gt;VELZARO&lt;/div&gt; &lt;/div&gt; &lt;div class="card-body"&gt; &lt;h2 class="card-title"&gt;${item.name}</h2>
<p class="notes">${item.notes}&lt;/p&gt; &lt;div class="card-footer"&gt; &lt;span class="price"&gt;${item.price.toLocaleString()} ج.م</span>
<button class="add-btn" onclick="addToCart(${item.id})"&gt;أضف للسلة&lt;/button&gt; &lt;/div&gt; &lt;/div&gt; &lt;/article&gt;).join("");
}

function addToCart(id) {
const existing = cart.find(item => item.id === id);
if (existing) {
existing.qty += 1;
} else {
const product = perfumes.find(p => p.id === id);
cart.push({ ...product, qty: 1 });
}
saveCart();
renderCart();
showToast("تمت إضافة العطر إلى سلة VELZARO");
}

function updateQty(id, delta) {
const item = cart.find(i => i.id === id);
if (!item) return;

item.qty += delta;
if (item.qty <= 0) {
cart = cart.filter(i => i.id !== id);
}
saveCart();
renderCart();
}

function renderCart() {
const totalItems = cart.reduce((sum, item) => sum + item.qty, 0);
const totalPrice = cart.reduce((sum, item) => sum + item.price * item.qty, 0);

cartCount.textContent = totalItems;
cartTotal.textContent = ${totalPrice.toLocaleString()} ج.م`;
checkoutBtn.disabled = cart.length === 0;

if (cart.length === 0) {
cartItemsContainer.innerHTML = <p style="color:#a1a1aa; text-align:center; margin-top:40px;"&gt;السلة فارغة حالياً.&lt;/p>;
return;
}

cartItemsContainer.innerHTML = cart.map(item => &lt;div class="cart-item"&gt; &lt;div&gt; &lt;strong&gt;${item.name}</strong>
<div style="color:#d4af37; font-size:0.9rem;">{item.id}, -1)">-</button>
<span>{item.id}, 1)">+</button>
</div>
</div>
`).join("");
}

document.getElementById("closeCartBtn").addEventListener("click", () => toggleDrawer(false));
overlay.addEventListener("click", () => toggleDrawer(false));

});
});
